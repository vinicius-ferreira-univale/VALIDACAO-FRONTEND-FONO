CREATE TABLE usuarios (
  id_usuario INT AUTO_INCREMENT PRIMARY KEY,
  nome VARCHAR(100) NOT NULL,
  email VARCHAR(100) NOT NULL,
  senha_hash VARCHAR(255) NOT NULL,
  perfil VARCHAR(20) NOT NULL,
  situacao VARCHAR(20) NOT NULL DEFAULT 'ATIVO',
  data_criacao DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

  CONSTRAINT uk_usuarios_email UNIQUE (email),

  CONSTRAINT chk_usuarios_perfil
    CHECK (perfil IN ('ADMIN', 'SECRETARIA', 'PROFESSOR', 'ALUNO', 'COORDENADOR')),

  CONSTRAINT chk_usuarios_situacao
    CHECK (situacao IN ('ATIVO', 'INATIVO', 'BLOQUEADO'))
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci;


CREATE TABLE responsaveis (
  id_responsavel INT AUTO_INCREMENT PRIMARY KEY,
  nome_completo VARCHAR(100) NOT NULL,
  cpf CHAR(11) NULL,
  whatsapp VARCHAR(20) NULL,
  situacao VARCHAR(20) NOT NULL DEFAULT 'ATIVO',

  CONSTRAINT uk_responsaveis_cpf UNIQUE (cpf),

  CONSTRAINT chk_responsaveis_cpf
    CHECK (cpf IS NULL OR CHAR_LENGTH(cpf) = 11),

  CONSTRAINT chk_responsaveis_situacao
    CHECK (situacao IN ('ATIVO', 'INATIVO'))
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci;


CREATE TABLE pacientes (
  id_paciente INT AUTO_INCREMENT PRIMARY KEY,

  situacao VARCHAR(20) NOT NULL DEFAULT 'ATIVO',

  CONSTRAINT uk_pacientes_cod_prontuario UNIQUE (cod_prontuario),
  CONSTRAINT uk_pacientes_cpf UNIQUE (cpf),

  CONSTRAINT chk_pacientes_cpf
    CHECK (cpf IS NULL OR CHAR_LENGTH(cpf) = 11),

  CONSTRAINT chk_pacientes_situacao
    CHECK (situacao IN ('ATIVO', 'INATIVO', 'ALTA', 'ARQUIVADO'))
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci;


CREATE TABLE periodo_letivo (
  id_periodo_letivo INT AUTO_INCREMENT PRIMARY KEY,
  periodo VARCHAR(20) NOT NULL,
  data_inicial DATE NOT NULL,
  data_final DATE NOT NULL,
  situacao VARCHAR(20) NOT NULL DEFAULT 'ATIVO',

  CONSTRAINT uk_periodo_letivo_periodo UNIQUE (periodo),

  CONSTRAINT chk_periodo_letivo_datas
    CHECK (data_final >= data_inicial),

  CONSTRAINT chk_periodo_letivo_situacao
    CHECK (situacao IN ('ATIVO', 'ENCERRADO', 'CANCELADO'))
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci;

CREATE TABLE responsaveis_pacientes (
  id_responsavel_paciente INT AUTO_INCREMENT PRIMARY KEY,
  id_responsavel INT NOT NULL,
  id_paciente INT NOT NULL,
  grau_parentesco VARCHAR(50) NULL,
  observacoes VARCHAR(255) NULL,

  CONSTRAINT uk_responsaveis_pacientes
    UNIQUE (id_responsavel, id_paciente),

  INDEX idx_resp_pac_paciente (id_paciente),

  CONSTRAINT fk_resp_pac_pacientes

    REFERENCES pacientes (id_paciente)
    ON DELETE CASCADE
    ON UPDATE CASCADE

  COLLATE=utf8mb4_unicode_ci;


CREATE TABLE matricula (
  id_matricula INT AUTO_INCREMENT PRIMARY KEY,
  id_grupo INT NOT NULL,
  id_aluno INT NOT NULL,
  id_periodo_letivo INT NOT NULL,
  data_matricula_inicio DATE NOT NULL,
  data_matricula_final DATE NULL,
  situacao VARCHAR(20) NOT NULL DEFAULT 'ATIVA',

  CONSTRAINT uk_matricula_grupo_aluno_periodo
    UNIQUE (id_grupo, id_aluno, id_periodo_letivo),

  INDEX idx_matricula_grupo (id_grupo),
  INDEX idx_matricula_aluno (id_aluno),
  INDEX idx_matricula_periodo_letivo (id_periodo_letivo),

  CONSTRAINT fk_matricula_grupo
    FOREIGN KEY (id_grupo)
    REFERENCES grupo (id_grupo)
    ON DELETE RESTRICT
    ON UPDATE CASCADE,

  CONSTRAINT fk_matricula_alunos
    FOREIGN KEY (id_aluno)
    REFERENCES alunos (id_aluno)
    ON DELETE RESTRICT
    ON UPDATE CASCADE,

  CONSTRAINT fk_matricula_periodo_letivo
    FOREIGN KEY (id_periodo_letivo)
    REFERENCES periodo_letivo (id_periodo_letivo)
    ON DELETE RESTRICT
    ON UPDATE CASCADE,

  CONSTRAINT chk_matricula_datas
    CHECK (
      data_matricula_final IS NULL
      OR data_matricula_final >= data_matricula_inicio
    ),

  CONSTRAINT chk_matricula_situacao
    CHECK (situacao IN ('ATIVA', 'ENCERRADA', 'CANCELADA', 'TRANCADA'))
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_unicode_ci;
